  Board      Arduino Uno R3 (ATmega328P, 16 MHz)
  Library    HX711 (only external library)
  Pins       HX711  DT=A1  SCK=A0
             Pump   D8   (relay, active HIGH)
             74HC595 DS=D11 SHCP=D12 STCP=D13
             Display digit select  A2 A3 A4 A5
             Keypad rows D2-D5, cols D6 D7 D9 D10

  What it does
    1. Reads the scale at 10 Hz through HX711, software tare.
    2. Waits for rice weight, multiplies by the ratio to get a target.
    3. Runs the pump, checks weight every 200 ms, stops EARLY by LEAD_G
       grams to compensate for the water still in the tube.
    4. Four guards fire on the way: timeout, overload, dry-run, on-target.
    5. Drives a 4-digit common-cathode display through a 74HC595.

  This file is a listing, not the build file.
  To compile: open demo_code.ino in the Arduino IDE and press Upload.

 249 | void loop() {
 250 |   static uint32_t tDisp = 0, tKey = 0, tIdle = 0;
 251 | 
 252 |   if (millis() - tDisp >= 2) { tDisp = millis(); scanDisplay(); }
 253 | 
 254 |   if (millis() - tKey >= 20) {
 255 |     tKey = millis();
 256 |     uint8_t k = kpScan();
 257 |     if (k != 255) handleKey(k);
 258 |   }
 259 | 
 260 |   pollSerial();
 261 | 
 262 |   if (running) {
 263 |     uint32_t now = millis();
 264 |     if (now - tSample >= SAMPLE_MS) {
 265 |       tSample = now;
 266 |       float w = readG();
 267 |       if (!isnan(w)) {
 268 |         showMilligrams((uint32_t)(w * 10 + 0.5));
 269 |         float added = w - riceW;
 270 | 
 271 |         if (now - t0 > TIMEOUT_MS)            { stopCycle(F("超时保护")); return; }
 272 |         if (w > MAX_TOTAL)                    { stopCycle(F("超重保护")); return; }
 273 |         if (now - t0 > 3000 && added < 0.5)   { stopCycle(F("没出水，查泵和管路")); return; }
 274 |         if (added >= target - LEAD_G)         { stopCycle(F("到量")); }
 275 |       }
 276 |     }
 277 |   } else {
 278 |     if (millis() - tIdle >= 300) {
 279 |       tIdle = millis();
 280 |       float w = readG();
 281 |       if (!isnan(w)) showMilligrams((uint32_t)(w * 10 + 0.5));
 282 |     }
 283 |   }
 284 | }
